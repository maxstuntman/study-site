wwindow.addEventListener('scroll', function() {
    const scroll = window.pageYOffset;
    
    // Слой 1 (Стол): движется очень медленно (далекий фон)
    const layer1 = document.querySelector('.layer-1');
    if (layer1) {
        layer1.style.transform = `translateY(${scroll * 0.15}px)`;
    }
    
    // Слой 2 (Руки): средняя скорость
    const layer2 = document.querySelector('.layer-2-hands');
    if (layer2) {
        layer2.style.transform = `translateY(${scroll * 0.4}px)`;
    }
    
    // Слой 3 (Кнопки): быстрее всех (ближний план)
    const layer3 = document.querySelector('.layer-3-content');
    if (layer3) {
        layer3.style.transform = `translateY(${50 + scroll * 0.7}px)`;
    }
});
